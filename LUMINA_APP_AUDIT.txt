import AsyncStorage from "@react-native-async-storage/async-storage";
import * as Location from "expo-location";
import { router, useLocalSearchParams } from "expo-router";
import { useMemo, useState } from "react";
import {
  ActivityIndicator,
  Alert,
  Linking,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import AddressMap from "@/components/AddressMap";
import { useAppTheme } from "@/providers/app-theme-provider";
import {
  isGeocodingConfigured,
  searchAddress,
  type GeocodeResult,
  type LocationPoint,
} from "@/lib/location-search";

const MAP_SELECTION_KEY = "lumina_map_selection_v1";

export default function AddressMapScreen() {
  const { colors } = useAppTheme();
  const params = useLocalSearchParams<{ latitude?: string; longitude?: string }>();
  const latitude = Array.isArray(params.latitude) ? params.latitude[0] : params.latitude;
  const longitude = Array.isArray(params.longitude) ? params.longitude[0] : params.longitude;
  const routePoint = useMemo(() => {
    if (!latitude || !longitude) return null;
    const lat = Number(latitude);
    const lon = Number(longitude);
    return Number.isFinite(lat) && Number.isFinite(lon)
      ? { latitude: lat, longitude: lon }
      : null;
  }, [latitude, longitude]);
  const [selectedPoint, setSelectedPoint] = useState<LocationPoint | null>(null);
  const point = selectedPoint ?? routePoint;
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<GeocodeResult[]>([]);
  const [busy, setBusy] = useState(false);
  const [searchAttempted, setSearchAttempted] = useState(false);
  const [error, setError] = useState("");

  const locate = async () => {
    setBusy(true);
    setError("");
    try {
      let permission = await Location.getForegroundPermissionsAsync();
      if (permission.status !== "granted" && permission.canAskAgain) {
        permission = await Location.requestForegroundPermissionsAsync();
      }
      if (permission.status !== "granted") {
        setError("Location access was not granted. You can still search or enter the full address manually.");
        if (!permission.canAskAgain) {
          Alert.alert(
            "Location permission is off",
            "Enable location in system settings if you want to place the pickup pin using your current location.",
            [
              { text: "Not now", style: "cancel" },
              { text: "Open settings", onPress: () => void Linking.openSettings() },
            ],
          );
        }
        return;
      }
      const location = await Location.getCurrentPositionAsync({
        accuracy: Location.Accuracy.Balanced,
      });
      setSelectedPoint({
        latitude: location.coords.latitude,
        longitude: location.coords.longitude,
      });
    } catch (locationError) {
      console.error("Could not retrieve current location:", locationError);
      setError("Current location is unavailable. Check your device location settings and try again, or enter the address manually.");
    } finally {
      setBusy(false);
    }
  };

  const explainAndLocate = () => {
    Alert.alert(
      "Use your current location?",
      "Lumina uses your location only to place a pickup pin for the address you choose. Your location is not collected in the background. You can enter the address manually instead.",
      [
        { text: "Enter manually", style: "cancel" },
        { text: "Continue", onPress: () => void locate() },
      ],
    );
  };

  const findAddress = async () => {
    setBusy(true);
    setError("");
    setSearchAttempted(true);
    try {
      const matches = await searchAddress(query);
      setResults(matches);
      if (matches.length === 0) setError("No matching addresses were returned. Check the spelling or enter the address manually.");
    } catch (searchError) {
      console.error("Address search failed:", searchError);
      setError(searchError instanceof Error ? searchError.message : "Address search failed. Try again or enter the address manually.");
      setResults([]);
    } finally {
      setBusy(false);
    }
  };

  const confirm = async () => {
    if (!point) return;
    try {
      const selectedResult = results.find(
        (result) =>
          result.latitude === point.latitude &&
          result.longitude === point.longitude,
      );
      await AsyncStorage.setItem(
        MAP_SELECTION_KEY,
        JSON.stringify({ point, label: selectedResult?.label ?? "" }),
      );
      router.back();
    } catch (saveError) {
      console.error("Map selection could not be saved:", saveError);
      setError("The selected map location could not be saved. Please try again.");
    }
  };

  const buttonStyle = {
    backgroundColor: colors.primary,
    borderColor: colors.primary,
  };

  return (
    <SafeAreaView style={[styles.safe, { backgroundColor: colors.background }]} edges={["top"]}>
      <ScrollView contentContainerStyle={styles.content} keyboardShouldPersistTaps="handled">
        <Pressable onPress={() => router.back()} accessibilityRole="button">
          <Text style={[styles.back, { color: colors.primary }]}>←  Back to address</Text>
        </Pressable>
        <Text style={[styles.eyebrow, { color: colors.primary }]}>PICKUP LOCATION</Text>
        <Text style={[styles.title, { color: colors.text }]}>Place a map pin</Text>
        <Text style={[styles.subtitle, { color: colors.textSecondary }]}>
          Confirm the address details separately. A pin does not verify address accuracy or pickup serviceability.
        </Text>

        <Pressable
          accessibilityRole="button"
          disabled={busy}
          onPress={explainAndLocate}
          style={[styles.actionButton, buttonStyle, busy && styles.disabled]}
        >
          <Text style={[styles.actionText, { color: colors.onPrimary }]}>
            ⌖  Use current location
          </Text>
        </Pressable>

        <View style={[styles.searchPanel, { backgroundColor: colors.backgroundElement, borderColor: colors.border }]}>
          <Text style={[styles.label, { color: colors.text }]}>Search an address</Text>
          <TextInput
            value={query}
            onChangeText={setQuery}
            placeholder={isGeocodingConfigured ? "Street, locality, or PIN code" : "Address search is not configured"}
            placeholderTextColor={colors.textSecondary}
            editable={isGeocodingConfigured && !busy}
            style={[styles.input, { backgroundColor: colors.input, borderColor: colors.border, color: colors.text }]}
            returnKeyType="search"
            onSubmitEditing={() => void findAddress()}
          />
          <Pressable
            accessibilityRole="button"
            disabled={!isGeocodingConfigured || busy || !query.trim()}
            onPress={() => void findAddress()}
            style={[styles.searchButton, { borderColor: colors.border }, (!isGeocodingConfigured || busy || !query.trim()) && styles.disabled]}
          >
            <Text style={[styles.searchText, { color: colors.primary }]}>
              {busy ? "Working…" : "Search"}
            </Text>
          </Pressable>
          {!isGeocodingConfigured && (
            <Text style={[styles.hint, { color: colors.textSecondary }]}>
              Configure EXPO_PUBLIC_GEOCODING_URL to enable search. Manual entry is always available.
            </Text>
          )}
        </View>

        {busy && <ActivityIndicator color={colors.primary} style={styles.loader} />}
        {!!error && (
          <View style={[styles.messageBox, { backgroundColor: colors.dangerBackground }]}>
            <Text style={{ color: colors.danger, fontSize: 12, lineHeight: 18 }}>{error}</Text>
            {searchAttempted && isGeocodingConfigured && (
              <Pressable onPress={() => void findAddress()} accessibilityRole="button" style={styles.retry}>
                <Text style={{ color: colors.danger, fontWeight: "800" }}>Retry search</Text>
              </Pressable>
            )}
          </View>
        )}
        {results.map((result) => {
          const selected =
            point?.latitude === result.latitude &&
            point.longitude === result.longitude;
          return (
            <Pressable
              key={`${result.latitude},${result.longitude}`}
              accessibilityRole="button"
              accessibilityState={{ selected }}
              onPress={() => setSelectedPoint(result)}
              style={[
                styles.result,
                {
                  backgroundColor: selected ? colors.backgroundSelected : colors.backgroundElement,
                  borderColor: selected ? colors.primary : colors.border,
                },
              ]}
            >
              <Text style={[styles.resultTitle, { color: colors.text }]}>{result.label}</Text>
              <Text style={[styles.hint, { color: colors.textSecondary }]}>
                {result.latitude.toFixed(5)}, {result.longitude.toFixed(5)}
              </Text>
            </Pressable>
          );
        })}

        <View style={styles.mapSection}>
          <AddressMap point={point} onPointChange={setSelectedPoint} />
          <Text style={[styles.hint, { color: colors.textSecondary }]}>
            {point
              ? `Selected pin: ${point.latitude.toFixed(5)}, ${point.longitude.toFixed(5)}. Tap the map to move it.`
              : "Search for a place or use your current location to place a pin. You may also skip this and enter the address manually."}
          </Text>
        </View>

        <Pressable
          accessibilityRole="button"
          disabled={!point || busy}
          onPress={() => void confirm()}
          style={[styles.actionButton, buttonStyle, (!point || busy) && styles.disabled]}
        >
          <Text style={[styles.actionText, { color: colors.onPrimary }]}>
            Confirm this pin
          </Text>
        </Pressable>
        <Pressable onPress={() => router.back()} accessibilityRole="button" style={styles.manualButton}>
          <Text style={[styles.manualText, { color: colors.primary }]}>Continue with manual address</Text>
        </Pressable>
      </ScrollView>
    </SafeAreaView>
  );
}

import AsyncStorage from "@react-native-async-storage/async-storage";
import { router, useFocusEffect, useLocalSearchParams } from "expo-router";
import { useCallback, useState } from "react";
import {
  ActivityIndicator,
  Alert,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import {
  deleteAddress,
  EMPTY_ADDRESS,
  formatAddress,
  getSelectedAddressId,
  loadAddresses,
  saveAddress,
  setDefaultAddress,
  setSelectedAddressId,
  type AddressLabel,
  type SavedAddress,
} from "@/lib/address-book";
import type { LocationPoint } from "@/lib/location-search";
import { useAppTheme } from "@/providers/app-theme-provider";
import { useThemedStyles } from "@/hooks/use-themed-styles";

const MAP_SELECTION_KEY = "lumina_map_selection_v1";

export default function AddressesScreen() {
  const { colors } = useAppTheme();
  const styles = useThemedStyles(baseStyles);
  const params = useLocalSearchParams<{ latitude?: string; longitude?: string }>();
  const latitude = Array.isArray(params.latitude) ? params.latitude[0] : params.latitude;
  const longitude = Array.isArray(params.longitude) ? params.longitude[0] : params.longitude;
  const initialCoordinates = (() => {
    if (!latitude || !longitude) return null;
    const lat = Number(latitude);
    const lon = Number(longitude);
    return Number.isFinite(lat) && Number.isFinite(lon)
      ? { latitude: lat, longitude: lon }
      : null;
  })();
  const [items, setItems] = useState<SavedAddress[]>([]);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [draft, setDraft] = useState<SavedAddress>(() => ({
    ...EMPTY_ADDRESS,
    coordinates: initialCoordinates,
  }));
  const [editing, setEditing] = useState(Boolean(initialCoordinates));
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const reload = useCallback(async (isActive: () => boolean = () => true) => {
    setLoading(true);
    setError("");
    try {
      const [addresses, selected] = await Promise.all([
        loadAddresses(),
        AsyncStorage.getItem("lumina_selected_address_v1"),
      ]);
      if (!isActive()) return;
      setItems(addresses);
      setSelectedId(selected);

      const mapSelection = await AsyncStorage.getItem(MAP_SELECTION_KEY);
      if (mapSelection) {
        let value: unknown;
        try {
          value = JSON.parse(mapSelection) as unknown;
        } catch {
          throw new Error("The selected map pin could not be read.");
        }
        if (value && typeof value === "object" && !Array.isArray(value)) {
          const selection = value as { point?: unknown; label?: unknown };
          if (selection.point && typeof selection.point === "object") {
            const point = selection.point as LocationPoint;
            if (
              typeof point.latitude === "number" &&
              Number.isFinite(point.latitude) &&
              typeof point.longitude === "number" &&
              Number.isFinite(point.longitude)
            ) {
              setDraft((current) => ({
                ...current,
                coordinates: point,
                street:
                  !current.street.trim() && typeof selection.label === "string"
                    ? selection.label
                    : current.street,
              }));
              setEditing(true);
            }
          }
        }
        await AsyncStorage.removeItem(MAP_SELECTION_KEY);
      }
    } catch (loadError) {
      console.error("Address book load failed:", loadError);
      if (isActive()) {
        setError(loadError instanceof Error ? loadError.message : "Could not load saved addresses.");
      }
    } finally {
      if (isActive()) setLoading(false);
    }
  }, []);

  useFocusEffect(
    useCallback(() => {
      let active = true;
      void reload(() => active);
      return () => {
        active = false;
      };
    }, [reload]),
  );

  const beginNew = () => {
    setDraft({ ...EMPTY_ADDRESS, label: items.length === 0 ? "Home" : "Other" });
    setEditing(true);
  };

  const save = async () => {
    setError("");
    try {
      const next = await saveAddress(draft);
      setItems(next);
      const saved = next.find((item) => item.id === (draft.id || next[0]?.id));
      if (saved && !selectedId) {
        await setSelectedAddressId(saved.id);
        setSelectedId(saved.id);
      }
      setDraft(EMPTY_ADDRESS);
      setEditing(false);
    } catch (saveError) {
      setError(saveError instanceof Error ? saveError.message : "Could not save this address.");
    }
  };

  const choose = async (item: SavedAddress) => {
    try {
      await setSelectedAddressId(item.id);
      setSelectedId(item.id);
      router.back();
    } catch (selectionError) {
      console.error("Address selection failed:", selectionError);
      setError(selectionError instanceof Error ? selectionError.message : "Could not select this address.");
    }
  };

  const remove = (item: SavedAddress) => {
    Alert.alert(
      "Remove saved address?",
      `Remove ${item.label === "Other" ? item.customLabel || "this address" : item.label}?`,
      [
        { text: "Cancel", style: "cancel" },
        {
          text: "Remove",
          style: "destructive",
          onPress: () => {
            void deleteAddress(item.id)
              .then(setItems)
              .then(() => {
                if (selectedId === item.id) {
                  return getSelectedAddressId().then(setSelectedId);
                }
              })
              .catch((removeError: unknown) => {
                Alert.alert(
                  "Address not removed",
                  removeError instanceof Error ? removeError.message : "Please try again.",
                );
              });
          },
        },
      ],
    );
  };

  const field = (
    key: keyof Pick<
      SavedAddress,
      | "customLabel"
      | "houseNumber"
      | "street"
      | "locality"
      | "landmark"
      | "city"
      | "state"
      | "pincode"
      | "phone"
      | "instructions"
    >,
    title: string,
    placeholder: string,
    options: { phone?: boolean; multiline?: boolean } = {},
  ) => (
    <View style={styles.field} key={key}>
      <Text style={[styles.label, { color: colors.text }]}>{title}</Text>
      <TextInput
        value={draft[key]}
        onChangeText={(value) =>
          setDraft((current) => ({
            ...current,
            [key]: options.phone ? value.replace(/\D/g, "") : value,
          }))
        }
        placeholder={placeholder}
        placeholderTextColor={colors.textSecondary}
        style={[
          styles.input,
          { backgroundColor: colors.input, borderColor: colors.border, color: colors.text },
          options.multiline && styles.multiline,
        ]}
        keyboardType={options.phone ? "phone-pad" : "default"}
        maxLength={options.phone ? (key === "phone" ? 10 : 6) : undefined}
        multiline={options.multiline}
      />
    </View>
  );

  return (
    <SafeAreaView style={[styles.safe, { backgroundColor: colors.background }]} edges={["top"]}>
      <ScrollView contentContainerStyle={styles.content} keyboardShouldPersistTaps="handled">
        <Pressable onPress={() => router.back()} accessibilityRole="button">
          <Text style={[styles.back, { color: colors.primary }]}>←  Back</Text>
        </Pressable>
        <Text style={[styles.eyebrow, { color: colors.primary }]}>PICKUP LOCATIONS</Text>
        <Text style={[styles.title, { color: colors.text }]}>Saved addresses</Text>
        <Text style={[styles.subtitle, { color: colors.textSecondary }]}>
          Select a pickup address or manage the addresses saved on this device.
        </Text>

        {!!error && (
          <View style={[styles.errorBox, { backgroundColor: colors.dangerBackground }]}>
            <Text style={{ color: colors.danger, fontSize: 12, lineHeight: 18 }}>{error}</Text>
          </View>
        )}

        {editing ? (
          <View style={[styles.card, { backgroundColor: colors.backgroundElement, borderColor: colors.border }]}>
            <Text style={[styles.section, { color: colors.text }]}>
              {draft.id ? "Edit address" : "Add address"}
            </Text>
            <Text style={[styles.label, styles.kindLabel, { color: colors.text }]}>Address label</Text>
            <View style={styles.chips}>
              {(["Home", "Work", "Other"] as AddressLabel[]).map((label) => (
                <Pressable
                  key={label}
                  accessibilityRole="radio"
                  accessibilityState={{ checked: draft.label === label }}
                  onPress={() => setDraft((current) => ({ ...current, label }))}
                  style={[
                    styles.chip,
                    {
                      backgroundColor: draft.label === label ? colors.backgroundSelected : colors.backgroundElement,
                      borderColor: draft.label === label ? colors.primary : colors.border,
                    },
                  ]}
                >
                  <Text style={[styles.chipText, { color: draft.label === label ? colors.primary : colors.text }]}>
                    {label}
                  </Text>
                </Pressable>
              ))}
            </View>
            {draft.label === "Other" && field("customLabel", "Custom label", "e.g. Studio")}
            {field("houseNumber", "House or flat number", "House 12, Flat 4B")}
            {field("street", "Street", "Street or road name")}
            {field("locality", "Locality", "Area or neighborhood")}
            {field("landmark", "Landmark (optional)", "Near a recognizable place")}
            <View style={styles.twoColumns}>
              <View style={styles.flex}>{field("city", "City", "City")}</View>
              <View style={styles.flex}>{field("state", "State", "State")}</View>
            </View>
            {field("pincode", "PIN code", "6-digit PIN code", { phone: true })}
            {field("phone", "Contact mobile (optional)", "10-digit mobile", { phone: true })}
            {field("instructions", "Pickup instructions (optional)", "Gate access or handover notes", { multiline: true })}
            <Text style={[styles.hint, { color: colors.textSecondary }]}>
              Address entry is manual. A map pin does not confirm serviceability.
            </Text>
            <Pressable
              onPress={async () => {
                try {
                  await AsyncStorage.removeItem(MAP_SELECTION_KEY);
                  router.push({
                    pathname: "/address-map",
                    params: {
                      latitude: draft.coordinates ? String(draft.coordinates.latitude) : "",
                      longitude: draft.coordinates ? String(draft.coordinates.longitude) : "",
                    },
                  });
                } catch (mapError) {
                  setError(mapError instanceof Error ? mapError.message : "Could not open map selection.");
                }
              }}
              style={[styles.mapButton, { borderColor: colors.primary }]}
            >
              <Text style={[styles.mapButtonText, { color: colors.primary }]}>
                {draft.coordinates ? "Adjust map pin" : "Choose map pin (optional)"}
              </Text>
            </Pressable>
            {!!draft.coordinates && (
              <Text style={[styles.hint, { color: colors.textSecondary }]}>
                Pin: {draft.coordinates.latitude.toFixed(5)}, {draft.coordinates.longitude.toFixed(5)}
              </Text>
            )}
            <Pressable onPress={() => void save()} style={[styles.primary, { backgroundColor: colors.primary }]}>
              <Text style={styles.primaryText}>Save address</Text>
            </Pressable>
            <Pressable onPress={() => { setDraft(EMPTY_ADDRESS); setEditing(false); }} style={styles.cancel}>
              <Text style={[styles.link, { color: colors.primary }]}>Cancel</Text>
            </Pressable>
          </View>
        ) : (
          <Pressable onPress={beginNew} style={[styles.addButton, { backgroundColor: colors.primary }]}>
            <Text style={styles.primaryText}>＋  Add new address</Text>
          </Pressable>
        )}

        {loading ? (

        ) : items.length === 0 && !editing ? (

                  <Text style={[styles.addressLabel, { color: colors.text }]}>

              </View>
              <Text style={[styles.address, { color: colors.textSecondary }]}>{formatAddress(item)}</Text>
              {!!item.landmark && (

              )}
              {!!item.phone && <Text style={[styles.meta, { color: colors.textSecondary }]}>Phone: {item.phone}</Text>}
              {!!item.instructions && (
                <Text style={[styles.meta, { color: colors.textSecondary }]}>Pickup note: {item.instructions}</Text>
              )}
              <View style={styles.actions}>
                <Pressable onPress={() => void choose(item)} style={[styles.primarySmall, { backgroundColor: colors.primary }]}>
                  <Text style={styles.primaryText}>Use this address</Text>
                </Pressable>
                <Pressable onPress={() => void setDefaultAddress(item.id).then((next) => {
                  setItems(next);
                  setSelectedId(item.id);
                }).catch((e: unknown) => setError(e instanceof Error ? e.message : "Could not update default address."))}>
                  <Text style={[styles.link, { color: colors.primary }]}>Set default</Text>
                </Pressable>
              </View>

  id: string;
  service_item_id: number;
  variant_id: number | null;
  vendor_id: number;
  name: string;
  price: number;
  quantity: number;
  unit: string;
};

function isCartItem(value: unknown): value is CartItem {
  if (!value || typeof value !== "object" || Array.isArray(value)) return false;
  const item = value as Record<string, unknown>;
  return (
    typeof item.id === "string" &&
    typeof item.service_item_id === "number" &&
    (typeof item.variant_id === "number" || item.variant_id === null) &&
    typeof item.vendor_id === "number" &&
    typeof item.name === "string" &&
    typeof item.price === "number" &&
    Number.isFinite(item.price) &&
    typeof item.quantity === "number" &&
    Number.isInteger(item.quantity) &&
    typeof item.unit === "string"
  );
}

function isCartItem(value: unknown): value is CartItem {
  if (!value || typeof value !== "object" || Array.isArray(value)) return false;
  const item = value as Record<string, unknown>;
  return (
    typeof item.id === "string" &&
    typeof item.service_item_id === "number" &&
    (typeof item.variant_id === "number" || item.variant_id === null) &&
    typeof item.vendor_id === "number" &&
    typeof item.name === "string" &&
    typeof item.price === "number" &&
    Number.isFinite(item.price) &&
    typeof item.quantity === "number" &&
    Number.isInteger(item.quantity) &&
    item.quantity > 0 &&
    typeof item.unit === "string"
  );
}

function toLocalDateString(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}

function formatSlot(slot: PickupSlot): string {
  const date = new Date(`${slot.date}T12:00:00`);
  const dateLabel = Number.isNaN(date.getTime())
    ? slot.date
    : date.toLocaleDateString("en-IN", {
        weekday: "short",
        day: "numeric",
        month: "short",
      });
  return `${dateLabel} · ${slot.startsAt}–${slot.endsAt}`;
}

function formatSavedAddressLabel(address: SavedAddress): string {
  return address.label === "Other"

    : address.label;

export default function FavoritesScreen(){
  const s = useThemedStyles(baseStyles);
  const router=useRouter();const [names,setNames]=useState<string[]>([]);
  const load=useCallback(()=>{let active=true;AsyncStorage.getItem("lumina_favorites").then(raw=>{if(!active)return;try{const value:unknown=raw?JSON.parse(raw):[];setNames(Array.isArray(value)?value.filter((x):x is string=>typeof x==="string"):[]);}catch{setNames([]);}});return()=>{active=false;};},[]);useFocusEffect(load);

import React, { useEffect, useRef, useState } from "react";
import {
  ActivityIndicator,
  Alert,
  Animated,
  Easing,
  KeyboardAvoidingView,
  Platform,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { router } from "expo-router";
import { signIn } from "@/lib/auth";

const AnimatedPressable = Animated.createAnimatedComponent(Pressable);

export default function LoginScreen() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [busy, setBusy] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const [emailFocused, setEmailFocused] = useState(false);
  const [passwordFocused, setPasswordFocused] = useState(false);

  const logoScale = useRef(new Animated.Value(0.7)).current;
  const logoOpacity = useRef(new Animated.Value(0)).current;
  const headingOpacity = useRef(new Animated.Value(0)).current;
  const headingY = useRef(new Animated.Value(22)).current;
  const formOpacity = useRef(new Animated.Value(0)).current;
  const formY = useRef(new Animated.Value(28)).current;
  const footerOpacity = useRef(new Animated.Value(0)).current;
  const orbRotation = useRef(new Animated.Value(0)).current;
  const buttonScale = useRef(new Animated.Value(1)).current;
  const emailBorder = useRef(new Animated.Value(0)).current;
  const passwordBorder = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    const entrance = Animated.sequence([
      Animated.parallel([
        Animated.timing(logoOpacity, {
          toValue: 1,
          duration: 450,
          useNativeDriver: true,
        }),
        Animated.spring(logoScale, {
          toValue: 1,
          damping: 12,
          stiffness: 120,
          useNativeDriver: true,
        }),
      ]),
      Animated.parallel([
        Animated.timing(headingOpacity, {
          toValue: 1,
          duration: 420,
          useNativeDriver: true,
        }),
        Animated.timing(headingY, {
          toValue: 0,
          duration: 520,
          easing: Easing.out(Easing.cubic),
          useNativeDriver: true,
        }),
      ]),
      Animated.parallel([
        Animated.timing(formOpacity, {
          toValue: 1,
          duration: 450,
          useNativeDriver: true,
        }),
        Animated.spring(formY, {
          toValue: 0,
          damping: 17,
          stiffness: 100,
          useNativeDriver: true,
        }),
        Animated.timing(footerOpacity, {
          toValue: 1,
          duration: 400,
          useNativeDriver: true,
        }),
      ]),
    ]);

    entrance.start();

    const floating = Animated.loop(
      Animated.sequence([
        Animated.timing(orbRotation, {
          toValue: 1,
          duration: 3200,
          easing: Easing.inOut(Easing.sin),
          useNativeDriver: true,
        }),
        Animated.timing(orbRotation, {
          toValue: 0,
          duration: 3200,
          easing: Easing.inOut(Easing.sin),
          useNativeDriver: true,
        }),
      ])
    );

    floating.start();

    return () => {
      entrance.stop();
      floating.stop();
    };
  }, [
    logoScale,
    logoOpacity,
    headingOpacity,
    headingY,
    formOpacity,
    formY,
    footerOpacity,
    orbRotation,
  ]);

  useEffect(() => {
    Animated.timing(emailBorder, {
      toValue: emailFocused ? 1 : 0,
      duration: 180,
      useNativeDriver: false,
    }).start();
  }, [emailFocused, emailBorder]);

  useEffect(() => {
    Animated.timing(passwordBorder, {
      toValue: passwordFocused ? 1 : 0,
      duration: 180,
      useNativeDriver: false,
    }).start();
  }, [passwordFocused, passwordBorder]);

  const login = async () => {
    const cleanEmail = email.trim();

    if (!/^\S+@\S+\.\S+$/.test(cleanEmail) || !password) {
      Alert.alert(
        "Check your details",
        "Enter a valid email address and password."
      );
      return;
    }

    setBusy(true);

    try {
      await signIn("customer", cleanEmail, password);
      router.replace("/(tabs)" as never);
    } catch (error) {
      Alert.alert(
        "Sign-in failed",
        error instanceof Error ? error.message : "Please try again."
      );
    } finally {
      setBusy(false);
    }
  };

  const orbTranslateY = orbRotation.interpolate({
    inputRange: [0, 1],
    outputRange: [0, -9],
  });

  const orbScale = orbRotation.interpolate({
    inputRange: [0, 0.5, 1],
    outputRange: [1, 1.04, 1],
  });

  const focusBorderColor = (value: Animated.Value) =>
    value.interpolate({
      inputRange: [0, 1],
      outputRange: ["#E3E6DC", "#648365"],
    });

  const pressIn = () => {
    Animated.spring(buttonScale, {
      toValue: 0.975,
      speed: 30,
      bounciness: 4,
      useNativeDriver: true,
    }).start();
  };

  const pressOut = () => {
    Animated.spring(buttonScale, {
      toValue: 1,
      speed: 22,
      bounciness: 5,
      useNativeDriver: true,
    }).start();
  };

  return (
    <SafeAreaView style={styles.safe} edges={["top", "bottom"]}>
      <KeyboardAvoidingView
        style={styles.keyboard}
        behavior={Platform.OS === "ios" ? "padding" : undefined}
      >
        <ScrollView
          contentContainerStyle={styles.content}
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}
        >
          <Animated.View
            style={[
              styles.brandRow,
              {
                opacity: logoOpacity,
                transform: [{ scale: logoScale }],
              },
            ]}
          >
            <View style={styles.logo}>
              <Text style={styles.logoText}>L</Text>
              <View style={styles.logoDot} />
            </View>

            <View>
              <Text style={styles.brand}>LUMINA</Text>
              <Text style={styles.brandSub}>LAUNDRY & CARE</Text>
            </View>

            <View style={styles.brandAccent}>
              <Text style={styles.brandAccentText}>✳</Text>
            </View>
          </Animated.View>

          <Animated.View
            style={[
              styles.heading,
              {
                opacity: headingOpacity,
                transform: [{ translateY: headingY }],
              },
            ]}
          >
            <View style={styles.eyebrowRow}>
              <View style={styles.accentLine} />
              <Text style={styles.eyebrow}>WELCOME BACK</Text>
            </View>

            <Text style={styles.title}>
              Your laundry,{"\n"}
              <Text style={styles.titleAccent}>made effortless.</Text>
            </Text>

            <Text style={styles.subtitle}>
              A little care goes a long way. Sign in to manage your orders and
              schedule your next pickup.
            </Text>

            <View style={styles.decorArea} pointerEvents="none">
              <Animated.View
                style={[
                  styles.decorOrb,
                  {
                    transform: [
                      { translateY: orbTranslateY },
                      { scale: orbScale },
                    ],
                  },
                ]}
              >
                <View style={styles.orbInner} />
              </Animated.View>
              <Text style={styles.decorSparkle}>✳</Text>

export default function OrderDetailsScreen() {
  const { id } = useLocalSearchParams<{ id?: string }>();
  const { colors } = useAppTheme();
  const [booking, setBooking] = useState<DemoBooking | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [rescheduleDate, setRescheduleDate] = useState("");
  const [rescheduleTime, setRescheduleTime] = useState("");
  const [notice, setNotice] = useState("");
  const [busy, setBusy] = useState(false);
  const currentStage = useMemo(() => booking?.status === "Cancelled" ? -1 : 0, [booking]);

  useEffect(() => {
    let active = true;
    loadDemoBookings().then((bookings) => {
      if (active) setBooking(bookings.find((item) => item.id === id) ?? null);
    }).catch((cause: unknown) => {
      if (active) setError(cause instanceof Error ? cause.message : "Could not load booking details.");
    }).finally(() => { if (active) setLoading(false); });
    return () => { active = false; };
  }, [id]);

  const saveRescheduleRequest = async () => {
    if (!booking || !/^\d{4}-\d{2}-\d{2}$/.test(rescheduleDate) || !rescheduleTime.trim()) {
      setNotice("Enter a date as YYYY-MM-DD and a preferred pickup time.");
      return;
    }
    setBusy(true);
    try {
      const raw = await AsyncStorage.getItem(RESCHEDULE_KEY);
      const existing: unknown = raw ? JSON.parse(raw) : [];
      if (!Array.isArray(existing)) throw new Error("Saved requests could not be read.");
      await AsyncStorage.setItem(RESCHEDULE_KEY, JSON.stringify([{ id: `LOCAL-${Date.now()}`, bookingId: booking.id, date: rescheduleDate, time: rescheduleTime.trim(), createdAt: new Date().toISOString() }, ...existing]));
      setNotice("Reschedule preference saved on this device. It has not been sent to a provider.");
      setRescheduleDate(""); setRescheduleTime("");
    } catch (cause) { setNotice(cause instanceof Error ? cause.message : "Could not save the request."); }
    finally { setBusy(false); }
  };

  const cancelLocally = async () => {
    if (!booking) return;
    setBusy(true);
    try {
      const updated = await cancelDemoBooking(booking.id);
      setBooking(updated.find((item) => item.id === booking.id) ?? null);
      setNotice("Local demo booking cancelled. No shop was notified.");
    } catch (cause) { setNotice(cause instanceof Error ? cause.message : "Could not update the local booking."); }
    finally { setBusy(false); }
  };

  return <SafeAreaView style={[styles.safe, { backgroundColor: colors.background }]} edges={["top"]}>

        <View style={[styles.card, { backgroundColor: colors.backgroundElement, borderColor: colors.border }]}>

import AsyncStorage from "@react-native-async-storage/async-storage";
import React, { useEffect, useState } from "react";
import {
  Alert,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  View,
} from "react-native";
import { router } from "expo-router";

type Draft = {
  owner: string;
  shop: string;
  phone: string;
  email: string;
  city: string;
};

const STORAGE_KEY = "lumina_vendor_registration_draft";
const EMPTY: Draft = { owner: "", shop: "", phone: "", email: "", city: "" };

export default function VendorRegisterScreen() {
  const [draft, setDraft] = useState<Draft>(EMPTY);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [storageError, setStorageError] = useState("");

  useEffect(() => {
    let active = true;
    AsyncStorage.getItem(STORAGE_KEY)
      .then((raw) => {
        if (!active || !raw) return;
        const value: unknown = JSON.parse(raw);
        if (!value || typeof value !== "object" || Array.isArray(value)) {
          throw new Error("Saved application draft has an invalid format.");
        }
        const saved = value as Partial<Draft>;
        setDraft({
          owner: typeof saved.owner === "string" ? saved.owner : "",
          shop: typeof saved.shop === "string" ? saved.shop : "",
          phone: typeof saved.phone === "string" ? saved.phone : "",
          email: typeof saved.email === "string" ? saved.email : "",
          city: typeof saved.city === "string" ? saved.city : "",
        });
      })
      .catch((error: unknown) => {
        if (active) {
          setStorageError(
            error instanceof Error
              ? error.message
              : "Saved application could not be loaded.",
          );
        }
      })
      .finally(() => {
        if (active) setLoading(false);
      });
    return () => {
      active = false;
    };
  }, []);

  const update = (key: keyof Draft, value: string) => {
    setDraft((current) => ({ ...current, [key]: value }));
  };

  const saveDraft = async () => {
    if (
      !draft.owner.trim() ||
      !draft.shop.trim() ||
      !/^[6-9]\d{9}$/.test(draft.phone.trim()) ||
      !/^\S+@\S+\.\S+$/.test(draft.email.trim()) ||
      !draft.city.trim()
    ) {
      Alert.alert(
        "Check your details",
        "Enter the owner's name, shop name, valid mobile number, business email and city.",
      );
      return;
    }
    setSaving(true);
    try {
      await AsyncStorage.setItem(
        STORAGE_KEY,
        JSON.stringify({
          ...draft,
          owner: draft.owner.trim(),
          shop: draft.shop.trim(),
          phone: draft.phone.trim(),
          email: draft.email.trim(),
          city: draft.city.trim(),
        }),
      );
      setStorageError("");
      Alert.alert(
        "Draft saved on this device",
        "Your shop application has not been submitted. Vendor onboarding needs the Lumina API.",
      );
    } catch (error) {
      const message =
        error instanceof Error ? error.message : "Please try again.";
      setStorageError(message);
      Alert.alert("Could not save draft", message);
    } finally {
      setSaving(false);
    }
  };

  const field = (
    label: string,
    key: keyof Draft,
    placeholder: string,
    phone = false,
  ) => (
    <View key={key} style={s.field}>
      <Text style={s.label}>{label}</Text>
      <TextInput
        value={draft[key]}
        onChangeText={(value) =>
          update(key, phone ? value.replace(/\D/g, "").slice(0, 10) : value)
        }
        placeholder={placeholder}
        placeholderTextColor="#4B5B50"
        keyboardType={phone ? "phone-pad" : key === "email" ? "email-address" : "default"}
        autoCapitalize={key === "email" ? "none" : "words"}
        maxLength={phone ? 10 : undefined}
        style={s.input}
      />
    </View>
  );

  return (
    <ScrollView
      style={s.page}
      contentContainerStyle={s.content}
      keyboardShouldPersistTaps="handled"
    >
      <Pressable onPress={() => router.back()}>
        <Text style={s.backText}>← Back</Text>
      </Pressable>
      <Text style={s.eyebrow}>JOIN THE LUMINA NETWORK</Text>
      <Text style={s.title}>Register your shop</Text>
      <Text style={s.description}>
        Add your business details to prepare a vendor onboarding application.
      </Text>
      <View style={s.card}>
        {field("Owner name", "owner", "Full name")}
        {field("Shop name", "shop", "Laundry shop name")}
        {field("Phone number", "phone", "10-digit mobile number", true)}
        {field("Business email", "email", "shop@example.com")}
        {field("City", "city", "Your city")}
        {!!storageError && <Text style={s.error}>{storageError}</Text>}
        <Pressable
          disabled={loading || saving}
          style={[s.primary, (loading || saving) && s.disabled]}
          onPress={() => void saveDraft()}
        >
          <Text style={s.primaryText}>
            {loading ? "Loading draft…" : saving ? "Saving…" : "Save application draft"}
          </Text>
        </Pressable>
        <Text style={s.note}>
          The draft stays on this device. Submitting it for review requires the
          vendor registration API.
        </Text>
      </View>
      <Pressable
        style={s.login}
        onPress={() => router.push("/vendor-login")}
      >
        <Text style={s.link}>Already a partner? Vendor sign in</Text>
      </Pressable>
    </ScrollView>
  );
}

function ThemedNavigation() {
  const { colors, effectiveMode } = useAppTheme();
  const [sessionReady, setSessionReady] = useState(false);

  useEffect(() => {
    let active = true;

    async function prepareFreshSession() {
      try {
        await Promise.all([
          clearAccessToken(),
          AsyncStorage.multiRemove([
            "lumina_profile",
            "lumina_mobile_session",
          ]),
        ]);
      } catch (error) {
        console.error("Could not clear the previous Lumina session:", error);
      } finally {
        if (active) setSessionReady(true);
      }
    }

    void prepareFreshSession();

    return () => {
      active = false;
    };
  }, []);

  if (!sessionReady) return null;

  return (
    <>
      <StatusBar style={effectiveMode === "dark" ? "light" : "dark"} />
      <Stack
        screenOptions={{
          headerShown: false,
          contentStyle: { backgroundColor: colors.background },

function getTrackingStage(status: string): number {
  const normalized = status.toLowerCase().replace(/[\s-]+/g, "_");

  if (["delivered", "completed"].includes(normalized)) return 5;

}

function parseProfile(value: unknown): ProfileDetails {
  if (!value || typeof value !== "object" || Array.isArray(value)) {
    return EMPTY_PROFILE;
  }

  const stored = value as Record<string, unknown>;
  return {
    name: typeof stored.name === "string" ? stored.name : "",
    phone: typeof stored.phone === "string" ? stored.phone : "",
    email: typeof stored.email === "string" ? stored.email : "",
    address: typeof stored.address === "string" ? stored.address : "",
  };
}

async function addToCart(service: CatalogService) {
  const raw = await AsyncStorage.getItem("lumina_cart");
  const cart: CartItem[] = raw ? JSON.parse(raw) : [];
  if (cart.length > 0 && cart.some(item => item.vendor_id !== service.vendor_id)) {
    throw new Error("Clear your cart before adding services from another shop.");
  }
  const variant = service.variants[0] ?? null;
  const cartId = `${service.id}-${variant?.id ?? "base"}`;
  const existing = cart.find(item => item.id === cartId);
  if (existing) existing.quantity = Math.min(50, existing.quantity + 1);
  else cart.push({
    id: cartId,
    service_item_id: service.id,
    variant_id: variant?.id ?? null,
    vendor_id: service.vendor_id,
    name: variant ? `${service.name} - ${variant.name}` : service.name,
    price: variant?.price ?? service.price,
    quantity: 1,
    unit: service.unit,
  });
  await AsyncStorage.setItem("lumina_cart", JSON.stringify(cart));
}
